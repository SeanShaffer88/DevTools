cmake_minimum_required(VERSION 3.20)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
option(DEVTOOLS_NO_CRT "Build the arithmetic-only sample without CRT/entrypoint" ON)
project(__NAME__ LANGUAGES C)
add_library(__NAME__ SHARED __NAME__.c)
target_include_directories(__NAME__ PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(__NAME__ PRIVATE DEVTOOLS_DLL_BUILD)
set_target_properties(__NAME__ PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
if(MSVC AND DEVTOOLS_NO_CRT)
    set(CMAKE_C_STANDARD_LIBRARIES "")
    string(REGEX REPLACE "/RTC[1su]" "" CMAKE_C_FLAGS_DEBUG "${CMAKE_C_FLAGS_DEBUG}")
    target_compile_options(__NAME__ PRIVATE /GS- /Zl)
    target_link_options(__NAME__ PRIVATE /NOENTRY /NODEFAULTLIB)
endif()
